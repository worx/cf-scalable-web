  --auto-scaling-group-name sandbox-php83-asg \
  --preferences '{"MinHealthyPercentage":50}' \
  --region us-east-1

  --auto-scaling-group-name sandbox-php83-asg \
  --query "InstanceRefreshes[0].{Status:Status,PctComplete:PercentageComplete}" \
